#include <stdio.h>
#include <string.h>
#include "avl_map.h"

// Funkcja porównująca stringi
int compare_strings(Key a, Key b) {
    return strcmp((char*)a, (char*)b);
}

// Funkcja dealokująca pamięć
void free_string_and_int(Key key, Value value) {
    free(key);      // dealokacja stringa
    free(value);    // dealokacja integera
}

// Callback do wyświetlania
void print_pair(Key key, Value value, void* user_data) {
    printf("Klucz: %s, Wartość: %d\n", (char*)key, *(int*)value);
}

int main() {
    // Tworzenie mapy
    AVLMap* map = avl_map_create(compare_strings, free_string_and_int);
    
    // Dodawanie elementów
    for (int i = 0; i < 10; i++) {
        char* key = (char*)malloc(20);
        int* value = (int*)malloc(sizeof(int));
        sprintf(key, "key%d", i);
        *value = i * 100;
        
        avl_map_insert(map, key, value);
    }
    
    // Wyszukiwanie
    char* search_key = "key5";
    int* found = (int*)avl_map_find(map, search_key);
    if (found) {
        printf("Znaleziono %s: %d\n", search_key, *found);
    }
    
    // Iteracja
    printf("\nWszystkie elementy:\n");
    avl_map_foreach(map, print_pair, NULL);
    
    // Usuwanie
    avl_map_remove(map, "key3");
    printf("\nPo usunięciu key3: %zu elementów\n", avl_map_size(map));
    
    // Czyszczenie
    avl_map_destroy(map);
    
    return 0;
}
